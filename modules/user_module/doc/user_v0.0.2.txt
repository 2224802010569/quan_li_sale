import 'package:core/di/injector.dart';
import 'package:core/di/supabase.dart';
import '../entity/user.dart';

class UserData {
  final String table = 'users';
  final supabase = get<SupabaseConnect>().client;

  final List<Map<String, dynamic>> localUserData = [
    {
      'id': 'Sale01',
      'password': '123',
      'phone': '0123456789',
      'role': 'Sale',
      'fullName': 'Nguyen Van A',
      'groupId': 'G1',
    },
    {
      'id': 'Mana01',
      'password': '123',
      'phone': '0123456789',
      'role': 'Manager',
      'fullName': 'Nguyen Van B',
      'groupId': 'G1',
    },
  ];

  User convertToUser(Map<String, dynamic> data) {
    return User(
      id: data['id'],
      password: data['password'],
      phone: data['phone'],
      role: data['role'],
      fullName: data['fullName'],
      groupId: data['groupId'],
    );
  }

  /// =========================
  /// CREATE (INSERT)
  /// =========================
  Future<bool> insertUser(User user) async {
    try {
      await supabase?.from(table).insert({
        'id': user.id,
        'password': user.password,
        'phone': user.phone,
        'role': user.role,
        'fullName': user.fullName,
        'groupId': user.groupId,
      });
      return true;
    } catch (error) {
      return false;
    }
  }

  /// =========================
  /// READ (GET ALL)
  /// =========================
  Future<List<User>> getAllUsers() async {
    try {
      final response = await supabase?.from(table).select();

      return response!.map<User>((item) => convertToUser(item)).toList();
    } catch (error) {
      /// fallback local
      return localUserData.map((item) => convertToUser(item)).toList();
    }
  }

  /// =========================
  /// UPDATE
  /// =========================
  Future<bool> updateUser(User user) async {
    try {
      await supabase
          ?.from(table)
          .update({
            'password': user.password,
            'phone': user.phone,
            'role': user.role,
            'fullName': user.fullName,
            'groupId': user.groupId,
          })
          .eq('id', user.id);

      return true;
    } catch (error) {
      return false;
    }
  }

  /// =========================
  /// DELETE (XÓA THẬT)
  /// =========================
  Future<bool> deleteUser(String id) async {
    try {
      await supabase?.from(table).delete().eq('id', id);
      return true;
    } catch (error) {
      return false;
    }
  }

  /// =========================
  /// LOGIN (XỬ LÝ RIÊNG)
  /// =========================
  Future<User?> login(String fullName, String password) async {
    try {
      final response = await supabase
          ?.from(table)
          .select()
          .eq('fullName', fullName)
          .maybeSingle();

      if (response != null && response['password'] == password) {
        return convertToUser(response);
      }
    } catch (error) {
      // ignore để fallback
    }

    /// fallback local
    for (var item in localUserData) {
      if (item['fullName'] == fullName && item['password'] == password) {
        return convertToUser(item);
      }
    }

    return null;
  }
}

class LoginUC {
  final _data = UserData();

  Future<User?> execute(String id, String password) async {
    if (id.isEmpty || password.isEmpty) {
      throw Exception("Thiếu thông tin");
    }

    final user = await _data.login(id, password);

    if (user != null) {
      AppStorage.saveUser({
        'id': user.id,
        'role': user.role,
        'groupId': user.groupId,
      });
    }

    return user;
  }
}

  final passController = TextEditingController();

        idController.text,
        passController.text,

        widget.onEvent(
          UserEvent.loginSuccess({
            'user_id': user.id,
            'role': user.role,
            'groupId': user.groupId,
          }),
        );

        setState(() => error = "Sai tài khoản hoặc mật khẩu");

      setState(() => error = e.toString());

  }

  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFF9F9F9),
      body: Center(
        child: SingleChildScrollView(
          child: ConstrainedBox(
            constraints: const BoxConstraints(maxWidth: 390),
